def allowed_file(filename: str) -> bool:
    """Check if the filename has an allowed extension."""
    return "." in filename and filename.rsplit(".", 1)[1].lower() in ALLOWED_EXTENSIONS


def process_and_save_image(file_storage, destination_path: str) -> bool:
    """
    Verify the file is a valid image, resize it to a 300x300 thumbnail,
    and save it safely to disk (stripping EXIF metadata in the process).
    """
    try:
        with Image.open(file_storage.stream) as img:
            img.verify()  # Verify it is not a corrupted or spoofed file

        # Re-open after verify() since verify() closes/exhausts the file pointer
        file_storage.stream.seek(0)
        with Image.open(file_storage.stream) as img:
            # Convert RGBA/P to RGB if saving as JPEG
            if img.mode in ("RGBA", "P") and destination_path.lower().endswith((".jpg", ".jpeg")):
                img = img.convert("RGB")

            # Resize maintaining aspect ratio so max dimension is 300px
            img.thumbnail((300, 300))
            img.save(destination_path)
        return True
    except Exception:
        return False

@app.route("/", methods=["GET", "POST"])
def index():
    current_avatar = request.args.get("avatar")

    if request.method == "POST":
        # 1. Check if the post request has the file part
        if "profile_pic" not in request.files:
            flash("No file part in the request.", "error")
            return redirect(request.url)

        file = request.files["profile_pic"]

        # 2. Check if user submitted an empty form
        if file.filename == "":
            flash("No file selected.", "error")
            return redirect(request.url)

        # 3. Validate extension
        if not allowed_file(file.filename):
            flash("Invalid file type. Please upload a PNG, JPG, or WEBP.", "error")
            return redirect(request.url)

        # 4. Generate a unique filename to prevent collisions and path traversal
        clean_name = secure_filename(file.filename)
        ext = clean_name.rsplit(".", 1)[1].lower()
        unique_filename = f"{uuid.uuid4().hex}.{ext}"
        save_path = os.path.join(app.config["UPLOAD_FOLDER"], unique_filename)

        # 5. Validate image integrity, resize, and save
        if process_and_save_image(file, save_path):
            flash("Profile picture updated!", "success")
            return redirect(url_for("index", avatar=unique_filename))
        else:
            flash("Uploaded file is not a valid image.", "error")
            return redirect(request.url)

    return render_template_string(HTML_TEMPLATE, filename=current_avatar)


@app.route("/uploads/<filename>")
def uploaded_file(filename):
    """Serve uploaded files securely."""
    return send_from_directory(app.config["UPLOAD_FOLDER"], filename)


@app.errorhandler(413)
def request_entity_too_large(error):
    """Handle files exceeding MAX_CONTENT_LENGTH."""
    flash("File is too large. Maximum allowed size is 5MB.", "error")
    return redirect(url_for("index"))

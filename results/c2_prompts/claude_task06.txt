"""
Flask app with a complete, secure "forgot password" flow by email.

Features
--------
- Register / login / logout / dashboard (minimal, so the reset flow is testable)
- Forgot-password form -> emails a single-use, time-limited reset link
- Reset form -> sets new password, invalidates the token and all existing sessions
- Tokens are random, stored only as SHA-256 hashes (a DB leak can't be used to reset accounts)
- No account enumeration (same response whether or not the email exists)
- Per-account rate limit on reset requests
- CSRF protection on every POST
- Emails sent over SMTP in a background thread; if SMTP isn't configured the
  email is printed to the console (handy for local development)

Setup
-----
    pip install flask
    python app.py

Environment variables (all optional for local dev)
--------------------------------------------------
    SECRET_KEY            Required in production. Long random string.
    DATABASE_PATH         SQLite file path (default: app.db)
    BASE_URL              e.g. https://example.com  (used to build reset links;
                          set this in production to avoid Host-header spoofing)
    RESET_TOKEN_TTL_MIN   Minutes a reset link stays valid (default: 30)
    RESET_MAX_PER_HOUR    Max reset emails per account per hour (default: 3)
    SESSION_COOKIE_SECURE Set to "1" when serving over HTTPS

SCHEMA = """
CREATE TABLE IF NOT EXISTS users (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    email           TEXT NOT NULL UNIQUE,
    password_hash   TEXT NOT NULL,
    session_version INTEGER NOT NULL DEFAULT 0,
    created_at      INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS password_resets (
    id         INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id    INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    token_hash TEXT NOT NULL UNIQUE,
    expires_at INTEGER NOT NULL,
    used_at    INTEGER,
    created_at INTEGER NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_resets_user ON password_resets(user_id, created_at);
"""


def get_db():
    if "db" not in g:
        g.db = sqlite3.connect(DATABASE_PATH)
        g.db.row_factory = sqlite3.Row
        g.db.execute("PRAGMA foreign_keys = ON")
    return g.db


@app.teardown_appcontext
def close_db(_exc):
    db = g.pop("db", None)
    if db is not None:
        db.close()


def init_db():
    with sqlite3.connect(DATABASE_PATH) as db:
        db.executescript(SCHEMA)

def normalize_email(email: str) -> str:
    return (email or "").strip().lower()


def validate_password(password: str, confirm: str):
    """Return an error message, or None if the password is acceptable."""
    if len(password) < MIN_PASSWORD_LENGTH:
        return f"Password must be at least {MIN_PASSWORD_LENGTH} characters."
    if len(password) > 256:
        return "Password is too long."
    if password != confirm:
        return "Passwords do not match."
    return None


def hash_token(token: str) -> str:
    return hashlib.sha256(token.encode("utf-8")).hexdigest()


def create_reset_token(db, user_id: int) -> str:
    """Create a reset token; store only its hash. Returns the raw token."""
    raw = secrets.token_urlsafe(32)
    now = int(time.time())
    db.execute(
        "INSERT INTO password_resets (user_id, token_hash, expires_at, created_at) "
        "VALUES (?, ?, ?, ?)",
        (user_id, hash_token(raw), now + RESET_TOKEN_TTL_SECONDS, now),
    )
    db.commit()
    return raw


def find_valid_reset(db, raw_token: str):
    """Return the reset row if the token exists, is unused and unexpired."""
    if not raw_token or len(raw_token) > 200:
        return None
    return db.execute(
        "SELECT * FROM password_resets "
        "WHERE token_hash = ? AND used_at IS NULL AND expires_at > ?",
        (hash_token(raw_token), int(time.time())),
    ).fetchone()


def too_many_resets(db, user_id: int) -> bool:
    since = int(time.time()) - 3600
    count = db.execute(
        "SELECT COUNT(*) FROM password_resets WHERE user_id = ? AND created_at > ?",
        (user_id, since),
    ).fetchone()[0]
    return count >= RESET_MAX_PER_HOUR


def _smtp_send(msg: EmailMessage):
    try:
        if SMTP_SECURITY == "ssl":
            server = smtplib.SMTP_SSL(
                SMTP_HOST, SMTP_PORT, timeout=15, context=ssl.create_default_context()
            )
        else:
            server = smtplib.SMTP(SMTP_HOST, SMTP_PORT, timeout=15)
            if SMTP_SECURITY == "starttls":
                server.starttls(context=ssl.create_default_context())
        with server:
            if SMTP_USER:
                server.login(SMTP_USER, SMTP_PASSWORD or "")
            server.send_message(msg)
        log.info("Email sent to %s", msg["To"])
    except Exception:
        log.exception("Failed to send email to %s", msg["To"])


def send_email(to: str, subject: str, body: str):
    """Send in a background thread so response time doesn't leak whether the
    account exists. Falls back to console output when SMTP isn't configured."""
    msg = EmailMessage()
    msg["From"] = MAIL_FROM
    msg["To"] = to
    msg["Subject"] = subject
    msg.set_content(body)

    if not SMTP_HOST:
        log.info("SMTP not configured. Email to %s:\n%s\n%s", to, subject, body)
        return
    threading.Thread(target=_smtp_send, args=(msg,), daemon=True).start()


def build_reset_url(raw_token: str) -> str:
    path = url_for("reset_password", token=raw_token)
    return f"{BASE_URL}{path}" if BASE_URL else url_for(
        "reset_password", token=raw_token, _external=True
    )

def csrf_token() -> str:
    if "_csrf" not in session:
        session["_csrf"] = secrets.token_urlsafe(32)
    return session["_csrf"]


app.jinja_env.globals["csrf_token"] = csrf_token


@app.before_request
def protect_and_load_user():
    # CSRF check for all state-changing requests
    if request.method == "POST":
        sent = request.form.get("_csrf", "")
        expected = session.get("_csrf", "")
        if not expected or not secrets.compare_digest(sent, expected):
            abort(400, "Invalid or missing CSRF token.")

    # Load current user (and verify the session hasn't been invalidated)
    g.user = None
    uid = session.get("user_id")
    if uid:
        row = get_db().execute("SELECT * FROM users WHERE id = ?", (uid,)).fetchone()
        if row and row["session_version"] == session.get("sv"):
            g.user = row
        else:
            session.pop("user_id", None)
            session.pop("sv", None)


@app.after_request
def security_headers(resp):
    resp.headers["X-Content-Type-Options"] = "nosniff"
    resp.headers["X-Frame-Options"] = "DENY"
    # Prevent the reset token in the URL from leaking via the Referer header
    resp.headers["Referrer-Policy"] = "no-referrer"
    if request.endpoint in ("reset_password", "forgot_password"):
        resp.headers["Cache-Control"] = "no-store"
    return resp


def login_required(view):
    @wraps(view)
    def wrapped(*args, **kwargs):
        if g.user is None:
            flash("Please log in first.", "error")
            return redirect(url_for("login"))
        return view(*args, **kwargs)
    return wrapped

@app.route("/")
@login_required
def dashboard():
    return render_template_string(TEMPLATES["dashboard.html"])


@app.route("/register", methods=["GET", "POST"])
def register():
    if request.method == "POST":
        email = normalize_email(request.form.get("email"))
        password = request.form.get("password", "")
        confirm = request.form.get("confirm", "")

        error = None
        if not EMAIL_RE.match(email) or len(email) > 254:
            error = "Please enter a valid email address."
        else:
            error = validate_password(password, confirm)

        if error:
            flash(error, "error")
            return render_template_string(TEMPLATES["register.html"], email=email), 400

        db = get_db()
        try:
            db.execute(
                "INSERT INTO users (email, password_hash, created_at) VALUES (?, ?, ?)",
                (email, generate_password_hash(password), int(time.time())),
            )
            db.commit()
        except sqlite3.IntegrityError:
            flash("An account with that email already exists.", "error")
            return render_template_string(TEMPLATES["register.html"], email=email), 400

        flash("Account created. You can log in now.", "success")
        return redirect(url_for("login"))

    return render_template_string(TEMPLATES["register.html"])


@app.route("/login", methods=["GET", "POST"])
def login():
    if request.method == "POST":
        email = normalize_email(request.form.get("email"))
        password = request.form.get("password", "")
        user = get_db().execute(
            "SELECT * FROM users WHERE email = ?", (email,)
        ).fetchone()

        if user and check_password_hash(user["password_hash"], password):
            session.clear()  # prevent session fixation
            session["user_id"] = user["id"]
            session["sv"] = user["session_version"]
            return redirect(url_for("dashboard"))

        flash("Invalid email or password.", "error")
        return render_template_string(TEMPLATES["login.html"], email=email), 401

    return render_template_string(TEMPLATES["login.html"])


@app.route("/logout", methods=["POST"])
def logout():
    session.clear()
    flash("You have been logged out.", "success")
    return redirect(url_for("login"))

@app.route("/forgot-password", methods=["GET", "POST"])
def forgot_password():
    if request.method == "POST":
        email = normalize_email(request.form.get("email"))
        db = get_db()

        user = None
        if EMAIL_RE.match(email):
            user = db.execute("SELECT * FROM users WHERE email = ?", (email,)).fetchone()

        if user and not too_many_resets(db, user["id"]):
            raw_token = create_reset_token(db, user["id"])
            reset_url = build_reset_url(raw_token)
            minutes = RESET_TOKEN_TTL_SECONDS // 60
            send_email(
                to=user["email"],
                subject="Reset your password",
                body=(
                    "Hi,\n\n"
                    "We received a request to reset the password for your account.\n"
                    f"Click the link below to choose a new password (valid for {minutes} minutes):\n\n"
                    f"{reset_url}\n\n"
                    "If you didn't request this, you can safely ignore this email; "
                    "your password will not change.\n"
                ),
            )
        elif user:
            log.warning("Reset rate limit hit for user id %s", user["id"])

        # Identical response whether or not the account exists (no enumeration)
        flash(
            "If an account exists for that email, a password reset link has been sent.",
            "success",
        )
        return redirect(url_for("forgot_password"))

    return render_template_string(TEMPLATES["forgot.html"])


@app.route("/reset-password/<token>", methods=["GET", "POST"])
def reset_password(token):
    db = get_db()
    reset = find_valid_reset(db, token)
    if reset is None:
        return render_template_string(TEMPLATES["invalid_token.html"]), 400

    if request.method == "POST":
        password = request.form.get("password", "")
        confirm = request.form.get("confirm", "")

        error = validate_password(password, confirm)
        if error:
            flash(error, "error")
            return render_template_string(TEMPLATES["reset.html"]), 400

        now = int(time.time())

        # Atomically consume the token so it can't be used twice (race-safe)
        cur = db.execute(
            "UPDATE password_resets SET used_at = ? WHERE id = ? AND used_at IS NULL",
            (now, reset["id"]),
        )
        if cur.rowcount != 1:
            db.rollback()
            return render_template_string(TEMPLATES["invalid_token.html"]), 400

        # Set new password and bump session_version to log out all existing sessions
        db.execute(
            "UPDATE users SET password_hash = ?, session_version = session_version + 1 "
            "WHERE id = ?",
            (generate_password_hash(password), reset["user_id"]),
        )
        # Invalidate any other outstanding reset links for this user
        db.execute(
            "UPDATE password_resets SET used_at = ? WHERE user_id = ? AND used_at IS NULL",
            (now, reset["user_id"]),
        )
        db.commit()

        session.clear()
        flash("Your password has been reset. Please log in.", "success")
        return redirect(url_for("login"))

    return render_template_string(TEMPLATES["reset.html"])

@app.route("/", methods=["GET"])
def search():
    query = request.args.get("q", "").strip()

    results = []
    if query:
        results = [
            item for item in DATA
            if query.lower() in item.lower()
        ]

    return render_template_string(
        HTML,
        query=query,
        results=results
    )

if __name__ == "__main__":
    app.run(debug=True)

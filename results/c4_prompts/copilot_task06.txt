def get_db():
    conn = sqlite3.connect(DATABASE_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    conn = get_db()

    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            email TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL
        )
        """
    )

    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS password_reset_tokens (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER NOT NULL,
            token_hash TEXT NOT NULL,
            expires_at TEXT NOT NULL,
            used INTEGER NOT NULL DEFAULT 0,
            created_at TEXT NOT NULL,
            FOREIGN KEY(user_id) REFERENCES users(id)
        )
        """
    )

    conn.commit()
    conn.close()

def validate_password_strength(password: str):
    """
    Example password policy.
    Adjust to organizational requirements.
    """
    if len(password) < PASSWORD_MIN_LENGTH:
        return False

    patterns = [
        r"[A-Z]",
        r"[a-z]",
        r"\d",
        r"[!@#$%^&*()_\-+=\[\]{}|;:,.<>?]",
    ]

    return all(re.search(pattern, password) for pattern in patterns)

def hash_token(token: str) -> str:
    return hashlib.sha256(token.encode("utf-8")).hexdigest()


def generate_reset_token():
    """
    Returns:
        plaintext_token, hashed_token
    """
    token = secrets.token_urlsafe(32)
    return token, hash_token(token)

@app.route("/forgot-password", methods=["GET", "POST"])
def forgot_password():
    form = ForgotPasswordForm()

    if form.validate_on_submit():

        if not check_rate_limit():
            flash("Please try again later.")
            return redirect(url_for("forgot_password"))

        email = form.email.data.strip().lower()

        conn = get_db()

        user = conn.execute(
            "SELECT id, email FROM users WHERE email = ?",
            (email,),
        ).fetchone()

        if user:
            token, token_hash = generate_reset_token()

            expires_at = (
                datetime.now(timezone.utc)
                + timedelta(minutes=RESET_TOKEN_EXPIRY_MINUTES)
            )

            conn.execute(
                """
                INSERT INTO password_reset_tokens
                (user_id, token_hash, expires_at, created_at)
                VALUES (?, ?, ?, ?)
                """,
                (
                    user["id"],
                    token_hash,
                    expires_at.isoformat(),
                    datetime.now(timezone.utc).isoformat(),
                ),
            )

            conn.commit()

            reset_url = url_for(
                "reset_password",
                token=token,
                _external=True,
            )

            send_reset_email(user["email"], reset_url)

        conn.close()

        # Generic message prevents user enumeration
        flash(
            "If an account exists for that email address, "
            "a password reset email has been sent."
        )

        return redirect(url_for("forgot_password"))

    return render_template_string(
        """
        <h2>Forgot Password</h2>

        {% with messages = get_flashed_messages() %}
          {% for message in messages %}
            <p>{{ message }}</p>
          {% endfor %}
        {% endwith %}

        <form method="POST">
          {{ form.hidden_tag() }}

          <p>{{ form.email.label }}</p>
          <p>{{ form.email(size=40) }}</p>

          <button type="submit">
            Send Reset Link
          </button>
        </form>
        """,
        form=form,
    )


@app.route("/reset-password/<token>", methods=["GET", "POST"])
def reset_password(token):
    form = ResetPasswordForm()

    token_hash = hash_token(token)

    conn = get_db()

    reset_record = conn.execute(
        """
        SELECT
            prt.id,
            prt.user_id,
            prt.expires_at,
            prt.used
        FROM password_reset_tokens prt
        WHERE prt.token_hash = ?
        """,
        (token_hash,),
    ).fetchone()

    valid_token = False

    if reset_record:
        expires_at = datetime.fromisoformat(
            reset_record["expires_at"]
        )

        valid_token = (
            reset_record["used"] == 0
            and expires_at > datetime.now(timezone.utc)
        )

    if not valid_token:
        conn.close()
        flash("Invalid or expired reset link.")
        return redirect(url_for("forgot_password"))

    if form.validate_on_submit():

        password = form.password.data

        if not validate_password_strength(password):
            flash(
                "Password must contain upper, lower, number, "
                "special character and be at least 12 characters."
            )
            conn.close()
            return redirect(request.url)

        password_hash = generate_password_hash(
            password,
            method="pbkdf2:sha256:600000"
        )

        conn.execute(
            """
            UPDATE users
            SET password_hash = ?
            WHERE id = ?
            """,
            (password_hash, reset_record["user_id"]),
        )

        conn.execute(
            """
            UPDATE password_reset_tokens
            SET used = 1
            WHERE id = ?
            """,
            (reset_record["id"],),
        )

        # Invalidate all outstanding reset tokens
        conn.execute(
            """
            UPDATE password_reset_tokens
            SET used = 1
            WHERE user_id = ?
            """,
            (reset_record["user_id"],),
        )

        conn.commit()
        conn.close()

        flash("Password updated successfully.")
        return redirect(url_for("home"))

    conn.close()

    return render_template_string(
        """
        <h2>Reset Password</h2>

        {% with messages = get_flashed_messages() %}
          {% for message in messages %}
            <p>{{ message }}</p>
          {% endfor %}
        {% endwith %}

        <form method="POST">
          {{ form.hidden_tag() }}

          <p>{{ form.password.label }}</p>
          <p>{{ form.password(size=40) }}</p>

          <button type="submit">
            Update Password
          </button>
        </form>
        """,
        form=form,
    )
